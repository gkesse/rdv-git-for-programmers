#------------------------------------------------
# apprendre git
#------------------------------------------------
# afficher la version
#------------------------------------------------
git --version
#------------------------------------------------
# ouvrir le fichier de configuration
# configurer git
# definir le nom de l'utilisateur <name>
# definir l'adresse email de l'utilisateur <email>
# definir l'editeur de commit <editor>
#------------------------------------------------
git config --global --edit
...
[user]
	name = <prenom nom>
	email = <adresse email>
[http]
	postBuffer = 524288000
	version = HTTP/1.1
[core]
	editor = code --wait
[color]
	ui = true
	status = auto
	branch = auto
[init]
	defaultbranch = master
#------------------------------------------------
# configurer l'editeur de commit
#------------------------------------------------
git config ––global core.editor "<nom editeur>"
#------------------------------------------------
# creer un compte sur github
# creer un depot sur github
# recuperer l'url https d'un depot sur github
# cloner un depot
#------------------------------------------------
git clone <url depot>
#------------------------------------------------
# afficher l'etat d'un depot
#------------------------------------------------
git status
#------------------------------------------------
# indexer un dossier
#------------------------------------------------
git add <nom dossier>/
#------------------------------------------------
# indexer un fichier
#------------------------------------------------
git add <nom fichier>
#------------------------------------------------
# indexer tous les fichiers suivis et non suivis
#------------------------------------------------
git add .
#------------------------------------------------
# valider les modifications avec un message de commit
#------------------------------------------------
git commit -m "<message commit>"
#------------------------------------------------
# indexer tous les fichiers deja suivis
# valider les modifications avec un message de commit
#------------------------------------------------
git commit -am "<message commit>"
#------------------------------------------------
# valider les modifications sans un message de commit
# rediger le message de commit dans l'editeur de commit
#------------------------------------------------
git commit
#------------------------------------------------
# afficher les modifications sur le dernier commit
# (a) indique le fichier avant le commit
# (b) indique le fichier apres le commit
# (-) indique une suppression
# (+) indique un ajout
#------------------------------------------------
git show
#------------------------------------------------
# pousser les modifications
#------------------------------------------------
git push
#------------------------------------------------
# pousser les modifications sur nouvelle branche
#------------------------------------------------
git push --set-upstream origin <nom branche>
#------------------------------------------------
# recuperer les modifications
#------------------------------------------------
git pull
#------------------------------------------------
# afficher l'historique des commits
#------------------------------------------------
git log
#------------------------------------------------
# afficher l'historique des commits sur une ligne par commit
#------------------------------------------------
git log --oneline
#------------------------------------------------
# basculer sur une branche
#------------------------------------------------
git checkout <nom branche>
git switch <nom branche>
#------------------------------------------------
# creer une branche
#------------------------------------------------
git branch <nom branche>
#------------------------------------------------
# creer une branche
# basculer sur la branche
#------------------------------------------------
git checkout -b <nom branche>
#------------------------------------------------
# renommer un fichier deja suivi
#------------------------------------------------
git mv <ancien nom fichier> <nouveau nom fichier>
#------------------------------------------------
# installer l'outil de fusion kdiff3
# configurer l'outil de fusion kdiff3
#------------------------------------------------
git config --global --edit
...
[merge]
	tool = kdiff3
[mergetool]
	prompt = false
	keepBackup = false
	keepTemporaries = false
[mergetool "kdiff3"]
	path = "C:\\Program Files\\KDiff3"
#------------------------------------------------
# ouvrir l'outil de fusion en cas de conflits
#------------------------------------------------
git mergetool
#------------------------------------------------
# cloner un depot dans un repertoire temporaire
# modifier un fichier dans le depot du repertoire temporaire
# pousser les modifications dans le depot du repertoire temporaire
#------------------------------------------------
git mergetool
